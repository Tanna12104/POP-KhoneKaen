ตั้งค่าเชื่อมต่อ Supabase Database

เพื่อให้คะแนน 1,262 โรงเรียนเชื่อมต่อฐานข้อมูลกลางเดียวกันและทำงานแบบเรียลไทม์ โปรดกรอก Supabase URL และ Anon Public Key ของคุณด้านล่าง:

คำสั่งสร้างตารางใน Supabase SQL Editor:

CREATE TABLE schools (
  id SERIAL PRIMARY KEY,
  name TEXT NOT NULL UNIQUE,
  score BIGINT DEFAULT 0,
  updated_at TIMESTAMPTZ DEFAULT NOW()
);
alter publication supabase_realtime add table schools;
